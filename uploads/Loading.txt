local script = G2L["8"];
	local fadeCycleTime = 1.7
	local turnCycleTime = 2
	
	local function spinnerEasingFunc(a, b, t)
		t = t * 2
	
		if t < 1 then
			return b / 2 * t*t*t + a
		else
			t = t - 2
			return b / 2 * (t * t * t + 2) + b
		end
	end
	
	game.RunService.RenderStepped:Connect(function(dt)
		local currentTime = tick()
	
		local timeInCycle = currentTime % turnCycleTime
		local cycleAlpha = spinnerEasingFunc(
			0,
			1,
			timeInCycle / turnCycleTime
		)
	
		script.Parent.Rotation = cycleAlpha * 360
	end)
end;
